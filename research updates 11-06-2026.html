## function to assign season given date
## seasons defined as epiweek 40 to epiweek 39 (roughly October to October)
find_season <- function(date) {
  date <- as.Date(date)
  epiyr <- epiyear(date)
  epiwk <- epiweek(date)
  start_yr <- ifelse(epiwk >= 40, epiyr, epiyr - 1)
  end_yr <- start_yr + 1
  paste0(start_yr,"-",end_yr)
}

However, just looking at the absolute minima and maxima indicated nothing about local peaks or the shape/pattern of the curve. I later tried using the findpeaks() function in the pracma package on the raw data, which ended up picking up on too much noise in the dataset. Here, I tried using LOESS regression (by state and flu season) to smooth out the noise, setting an arbitrary span of \(\alpha = 0.4\).

Using the values predicted by the LOESS model on our data, I used the findpeaks() function, with arbitrary arguments minpeakdistance = 4, threshold = 1 to identify relative peaks by state/season, storing them all in the tibble state_peaks.

Code 
# inputs for data and the specific season 
get_peaks <- function(group_data, season_input) {
  # using arbitrary threshold (weekly rate > 1, distance between peaks
  # required to be > 4 weeks)
  initial_peaks <- findpeaks(group_data, minpeakdistance = 4, threshold = 1)
  peaks <- if (is.null(initial_peaks)) {matrix(data = NA_real_, ncol = 4)} 
  else {initial_peaks}
  
  dates_in_order <- flu_data |>
    filter(season == season_input) |>
    select(date) |> unique() |> as.matrix() |> sort()
  
  peaks[,2] <- dates_in_order[as.numeric(peaks[,2])]
  peaks[,3] <- dates_in_order[as.numeric(peaks[,3])]
  peaks[,4] <- dates_in_order[as.numeric(peaks[,4])]
  
  return(peaks)
}

# dataset with relative peaks using findpeaks() on LOESS regression model
state_peaks <- flu_loess|>
  filter(!season %in% c("2021-2022", "2026-2027")) |>

In the visualizations below, you can see the peaks (with start and end dates) overlaid over trends with fairly clean data (Virginia) and noisy data (Wyoming). Using just findpeaks() with no smoothing resulted in four peaks for the Wyoming curve, whereas the LOESS regression smooths it to one main peak.

There are some irregularities, such as no peak in the 2023-2024 Alaska flu season, likely because it started so early that it overlapped with the 2022-2023 dates in the dataset.

Code 
peak_plot <- function (given_states, given_seasons) {
  flu_loess_filtered <- flu_loess |> filter(location_name %in% given_states, 
                                            season %in% given_seasons)
  state_peaks_filtered <- state_peaks |> filter(location_name == given_states, 
                                            season %in% given_seasons)
  
  plot <- ggplot(data = flu_loess_filtered, 
  mapping = aes(x = as.Date(date), y = weekly_rate)) +
    geom_point(size = 0.5) +
    geom_smooth(method = "loess", span = 0.4) +
    geom_rect(data = state_peaks_filtered, 
              mapping = aes(xmin = start_date, xmax = end_date, 
                            ymin = -Inf, ymax = Inf),
              inherit.aes = FALSE, fill = "red", alpha = 0.15) +
    theme_minimal() +
    labs(x = "Date", y = "Weekly Hospitalization Rate (per 100k)", title = paste0("Weekly Hospitalization Rates in ", given_states, ", ", paste(given_seasons, collapse = " and ")))
  return(plot)
}

other smoothing methods (11-06-2026)

Savitzky-Golay filter

Here, I used a Savitzky-Golay filter to identify peaks, which has two parameters: window size (number of weeks around a given datapoint used for approximation) and polynomial degree (degree of the polynomial fitted to the datapoints in said window). So far, I’ve set the parameters to 11 for window size and 3 for polynomial degree, which led to narrower peaks than the original LOESS model and identification of the peak at the beginning of the Alaska 2023-2024 flu season.

Code 
flu_sg <- flu_data |>
  arrange(location_name, date) |>
  group_by(location_name) |>
  mutate(sg_smoothed = savgol(weekly_rate, fl = 11, forder = 3, dorder = 0))

state_peaks_sg <- flu_sg |>
  filter(!season %in% c("2021-2022", "2026-2027")) |>
  group_by(location_name, season) |>
